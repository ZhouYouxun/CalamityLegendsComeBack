这篇报告旨在指导你如何在自己的 Terraria 模组中，重现类似于救赎模组（Mod of Redemption）中“盲目正义·恶魔之怖”武器所发射的**神圣光粒子追踪弹幕**（即原版的 `Lightmass` 弹幕）。

## 2. 核心特效拆解与模仿思路

要在你自己的模组中实现它，我们只需重新编写最核心的**光粒子追踪弹幕**。它的视觉美感来自以下三个设定：

1.  **节奏感 (Rhythm)**：
    *   刚刚发射时：速度乘算 `0.98` 进行阻尼减速，造成子弹“悬停在天空中”的静谧感。
    *   悬停结束时：检测并锁定最近的目标，以极高的转向速度冲刺。
2.  **发光与残影 (Glow & Afterimage)**：
    *   由于去掉了救赎专用的着色器（Shader）和顶点渲染（DanTrail），我们改用**原版残影缓存 (`oldPos`)**，配合叠加混合模式（`BlendState.Additive`），使用半透明的星光贴图绘制多重重叠的残影，效果完全可以以假乱真。
3.  **撞击缓冲 (Fake Kill)**：
    *   当子弹击中怪物或者墙壁时，它不会瞬间“蒸发”消失，而是让速度归零并隐形，在碰撞处生成一堆金色火焰粒子。此时由于弹幕还在，它原先产生的拖尾会自然收缩淡出，视觉上极其顺滑。

## 3. 极简 C# 特效弹幕代码模板

你可以直接将该代码作为弹幕文件导入到你的模组中。你需要准备一张类似于**十字星光**或**发光圆球**的白色发光贴图。

```csharp
using Microsoft.Xna.Framework;
using Microsoft.Xna.Framework.Graphics;
using System;
using Terraria;
using Terraria.GameContent;
using Terraria.ID;
using Terraria.ModLoader;

namespace CalamityLegendsComeBack.Effects
{
    public class BlindJusticeStar_Proj : ModProjectile
    {
        // 建议在你的模组贴图文件夹中放一张亮白色星光/耀斑贴图，并在此处填入路径
        public override string Texture => "CalamityLegendsComeBack/Effects/StarGlow";

        public override void SetStaticDefaults()
        {
            // 启用原版残影位置缓存，缓存 8 帧的位置
            ProjectileID.Sets.TrailCacheLength[Projectile.type] = 8;
            ProjectileID.Sets.TrailingMode[Projectile.type] = 0;
        }

        public override void SetDefaults()
        {
            Projectile.width = 16;
            Projectile.height = 16;
            Projectile.friendly = true;
            Projectile.hostile = false;
            Projectile.tileCollide = false; // 粒子可以穿墙以防追踪时卡住，由AI判断碰撞
            Projectile.penetrate = 1;
            Projectile.DamageType = DamageClass.Melee;
            Projectile.timeLeft = 180;
            
            // 随机粒子大小，让发射出来的星光更有参差感
            Projectile.scale = Main.rand.NextFloat(0.6f, 1.1f);
        }

        private bool isFakeKilling = false;

        public override void AI()
        {
            // 如果撞击后处于“伪死亡”虚化状态，则不执行任何运动
            if (isFakeKilling)
            {
                Projectile.velocity = Vector2.Zero;
                Projectile.alpha = 255; // 身体隐形，仅靠 PreDraw 绘制残影收缩
                return;
            }

            // 如果粒子卡在实体砖块里，加快其消失速度
            if (Collision.SolidCollision(Projectile.position, Projectile.width, Projectile.height))
            {
                Projectile.timeLeft -= 2;
            }

            // 核心AI分段控制：
            // 前 30 帧 (timeLeft > 150)：减速悬停阶段
            if (Projectile.timeLeft > 150)
            {
                Projectile.velocity *= 0.96f; // 阻尼减速
            }
            // 30 帧之后：锁定最近的目标冲刺
            else
            {
                NPC target = null;
                float maxDistance = 900f;

                // 寻找最近的可追踪敌人
                for (int i = 0; i < Main.maxNPCs; i++)
                {
                    NPC npc = Main.npc[i];
                    if (npc.active && !npc.friendly && npc.CanBeChasedBy())
                    {
                        float distanceTo = Vector2.Distance(Projectile.Center, npc.Center);
                        if (distanceTo < maxDistance)
                        {
                            maxDistance = distanceTo;
                            target = npc;
                        }
                    }
                }

                // 如果找到目标，转向并加速冲锋
                if (target != null)
                {
                    Vector2 desiredVelocity = (target.Center - Projectile.Center).SafeNormalize(Vector2.Zero) * 16f;
                    
                    // 渐进转向（插值），20 代表转向平滑度，数值越小转向越灵敏
                    Projectile.velocity = Vector2.Lerp(Projectile.velocity, desiredVelocity, 0.08f);
                }
                else
                {
                    // 没目标就保持缓慢漂浮
                    Projectile.velocity *= 0.98f;
                }
            }

            // 自转效果
            Projectile.rotation += 0.05f * Projectile.direction;
        }

        // 击中怪物时触发“伪死亡”
        public override void OnHitNPC(NPC target, NPC.HitInfo hit, int damageDone)
        {
            TriggerFakeKill();
        }

        // 碰撞检测（自己检测实体墙壁以防穿墙撞击失效）
        public override bool TileCollideStyle(ref int width, ref int height, ref bool fallThrough, ref Vector2 hitboxVelocity)
        {
            if (Collision.SolidCollision(Projectile.position, Projectile.width, Projectile.height))
            {
                TriggerFakeKill();
            }
            return false;
        }

        // 触发撞击爆炸虚化逻辑
        private void TriggerFakeKill()
        {
            if (isFakeKilling) return;
            isFakeKilling = true;
            
            Projectile.friendly = false;
            Projectile.timeLeft = 8; // 仅多留 8 帧让残影拖尾自然收缩

            // 在撞击点生成一圈爆开的发光金色粉尘
            for (int i = 0; i < 8; i++)
            {
                Vector2 speed = Main.rand.NextVector2Circular(4f, 4f);
                Dust dust = Dust.NewDustPerfect(Projectile.Center, DustID.GoldFlame, speed, Scale: 1.2f);
                dust.noGravity = true;
            }
        }

        // 核心特效实现：叠加模式绘制残影拖尾
        public override bool PreDraw(ref Color lightColor)
        {
            Texture2D texture = TextureAssets.Projectile[Projectile.type].Value;
            Vector2 origin = texture.Size() / 2f;

            // 切换为叠加混合模式 (Additive)，能让光粒子在背景上叠加显示出强烈的发光感
            Main.spriteBatch.End();
            Main.spriteBatch.Begin(SpriteSortMode.Deferred, BlendState.Additive, Main.DefaultSamplerState, DepthStencilState.None, RasterizerState.CullCounterClockwise, null, Main.GameViewMatrix.TransformationMatrix);

            // 1. 绘制历史残影拖尾
            for (int i = Projectile.oldPos.Length - 1; i >= 0; i--)
            {
                Vector2 drawPos = Projectile.oldPos[i] + Projectile.Size / 2f - Main.screenPosition;
                // 越老的位置，透明度越低
                float opacity = (Projectile.oldPos.Length - i) / (float)Projectile.oldPos.Length;
                
                // 渐变颜色：外圈偏金黄色，核心亮白色
                Color trailColor = Color.Lerp(Color.Gold, Color.White, opacity) * opacity * 0.8f;
                // 如果在伪死亡期间，逐渐淡出
                if (isFakeKilling) trailColor *= (Projectile.timeLeft / 8f);

                Main.EntitySpriteDraw(
                    texture,
                    drawPos,
                    null,
                    trailColor,
                    Projectile.rotation * (i * 0.2f), // 残影自带轻微扭转
                    origin,
                    Projectile.scale * (0.4f + opacity * 0.6f), // 拖尾逐渐收缩变细
                    SpriteEffects.None,
                    0
                );
            }

            // 2. 绘制弹幕主体本身
            if (!isFakeKilling)
            {
                Vector2 mainDrawPos = Projectile.Center - Main.screenPosition;
                Main.EntitySpriteDraw(
                    texture,
                    mainDrawPos,
                    null,
                    Color.White * 0.9f,
                    Projectile.rotation,
                    origin,
                    Projectile.scale,
                    SpriteEffects.None,
                    0
                );
            }

            // 恢复原版的默认混合模式
            Main.spriteBatch.End();
            Main.spriteBatch.Begin(SpriteSortMode.Deferred, BlendState.AlphaBlend, Main.DefaultSamplerState, DepthStencilState.None, RasterizerState.CullCounterClockwise, null, Main.GameViewMatrix.TransformationMatrix);

            return false; // 禁用原版普通的无发光绘制
        }
    }
}
```

## 4. 总结

这份设计通过巧妙的**帧数控制（30帧悬停、8帧伪死）**与**残影缩放**，在不写任何图形着色器的前提下，模拟出了救赎模组原版极高档的光晕滞空与冲刺效果。您的朋友只需更改 `StarGlow` 贴图的形态（如换成雪花、符文或火焰球），就可以瞬间将这个神圣星光改造成任何个性化的魔幻弹幕。
