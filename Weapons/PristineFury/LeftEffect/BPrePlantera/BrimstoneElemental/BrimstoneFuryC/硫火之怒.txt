   private int frameCounter = 0; // 用于计数帧数
        private int attackCycle = 0; // 当前攻击循环计数
        private const int FireInterval = 40; // 每轮攻击的冷却时间
        private const float BaseShootSpeed = 15f; // 基础射速

        public override void HoldoutAI()
        {
            frameCounter++;

            // 冷却阶段，等待冷却时间结束
            if (frameCounter < FireInterval)
                return;

            frameCounter = 0; // 重置计数器

            // 判断是否为激光射击阶段
            if (attackCycle >= 5)
            {
                FireLaser();
                attackCycle = 0; // 重置攻击循环
            }
            else
            {
                FireProjectiles();
                attackCycle++; // 增加攻击循环计数
            }
        }

        private void FireProjectiles()
        {
            Player player = Main.player[Projectile.owner];
            Vector2 shootDirection = Projectile.velocity.SafeNormalize(Vector2.UnitX * player.direction);

            Item heldItem = player.HeldItem;
            if (player.HasAmmo(heldItem))
            {
                if (player.PickAmmo(heldItem, out int ammoProjectile, out float shootSpeed, out int damage, out float knockback, out int ammoType))
                {
                    // 计算垂直向量
                    Vector2 perpendicular = shootDirection.RotatedBy(MathHelper.PiOver2);

                    // 发射弹幕
                    for (int i = -2; i <= 2; i++) // 一共5个弹幕
                    {
                        Vector2 offset = perpendicular * (8f * i); // 偏移量随索引变化
                        CreateProjectile(Projectile.Center + offset, shootDirection, ammoProjectile, damage, knockback, player);
                    }

                    // 播放音效
                    SoundEngine.PlaySound(SoundID.Item5, player.Center);
                }
            }
        }


        private void FireLaser()
        {
            Player player = Main.player[Projectile.owner];
            Vector2 shootDirection = Projectile.velocity.SafeNormalize(Vector2.UnitX * player.direction);

            // 发射激光
            int laserProjectile = Projectile.NewProjectile(
                Projectile.GetSource_FromThis(),
                Projectile.Center,
                shootDirection * BaseShootSpeed,
                ModContent.ProjectileType<BrimstoneFuryReBEAM>(), // 替换为实际激光弹幕类型
                Projectile.damage * 2, // 激光造成更高的伤害
                Projectile.knockBack,
                player.whoAmI
            );

            //// 修改激光的伤害类型为射手
            //if (laserProjectile.WithinBounds(Main.maxProjectiles))
            //{
            //    Projectile proj = Main.projectile[laserProjectile];
            //    proj.DamageType = DamageClass.Ranged; // 设置为射手伤害
            //}

            // 播放激光音效
            SoundEngine.PlaySound(SoundID.Item28, player.Center);
        }

        private void CreateProjectile(Vector2 position, Vector2 direction, int ammoProjectile, int damage, float knockback, Player player)
        {
            Projectile.NewProjectile(
                Projectile.GetSource_FromThis(),
                position,
                direction * BaseShootSpeed,
                ammoProjectile == ProjectileID.WoodenArrowFriendly ? ModContent.ProjectileType<BrimstoneFuryRePROJ>() : ammoProjectile,
                damage,
                knockback,
                player.whoAmI
            );
        }
